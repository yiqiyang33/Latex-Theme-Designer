\section{Theorem Family Blocks}

\defn{Metric Space}{def:metric-space}{
    A metric space is a pair $(X, d)$ where $d : X \times X \to \mathbb{R}_{\ge 0}$
    satisfies non-negativity, identity of indiscernibles, symmetry, and triangle inequality.
}

\thm{Bound from triangle inequality}{thm:triangle-bound}{
    For any $x,y,z \in X$ in a metric space $(X,d)$,
    \[
        d(x,z) \le d(x,y) + d(y,z).
    \]
}

\pf{
    This is one of the metric axioms, so no additional argument is needed.
}

\lem{Two-step upper bound}{lem:two-step}{
    If $d(x,y)\le a$ and $d(y,z)\le b$, then $d(x,z)\le a+b$.
}

\cor{Specialized estimate}{cor:special}{
    If $d(x,y)\le \varepsilon$ and $d(y,z)\le \varepsilon$, then $d(x,z)\le 2\varepsilon$.
}

\prop{Linearity of expectation}{prop:linearity}{
    For integrable random variables $X,Y$ and scalars $\alpha,\beta$,
    \[
        \mathbb{E}[\alpha X + \beta Y] = \alpha \mathbb{E}[X] + \beta \mathbb{E}[Y].
    \]
}

\clm{Simple monotonicity claim}{clm:mono}{
    If $a \le b$ and $c \ge 0$, then $ac \le bc$.
}

\fact{Algebraic identity}{fact:identity}{
    For real numbers $u,v$,
    \[
        u^2 - v^2 = (u-v)(u+v).
    \]
}

\assumption{Smooth objective}{assump:smooth}{
    The objective function $f$ is differentiable and its gradient is $L$-Lipschitz.
}
