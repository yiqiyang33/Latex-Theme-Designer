\sethlcolor{inline-term-bg}

\makeatletter
\NewDocumentCommand{\themeInlineBox}{O{} m m +m}{%
    \tcbox[
        on line,
        nobeforeafter,
        box align=base,
        boxsep=0pt,
        left=1.7pt,
        right=1.7pt,
        top=0.35pt,
        bottom=0.35pt,
        arc=1.6pt,
        outer arc=1.6pt,
        boxrule=0.25pt,
        colback=#2,
        colframe=#3!30!white,
        colupper=#3,
        #1
    ]{\strut #4}%
}
\newcommand{\key}[1]{\themeInlineBox[fontupper=\bfseries]{inline-term-bg}{inline-key-fg}{#1}}
\newcommand{\term}[1]{\themeInlineBox{inline-term-bg}{inline-term-fg}{#1}}
\newcommand{\warn}[1]{\themeInlineBox[fontupper=\bfseries]{inline-todo-bg}{inline-warn-fg}{#1}}
\newcommand{\todo}[1]{\themeInlineBox[fontupper=\sffamily\footnotesize\bfseries]{inline-todo-bg}{inline-todo-fg}{TODO\hspace{0.25em}#1}}
\newcommand{\code}[1]{%
    \themeInlineBox[fontupper=\ttfamily\footnotesize]{inline-code-bg}{inline-code-fg}{\detokenize{#1}}%
}
\newcommand{\sidenote}[1]{%
    \marginnote{%
        \footnotesize
        \setlength{\parindent}{0pt}%
        \color{sidenote-fg}%
        \textcolor{sidenote-accent}{\rule{1.3pt}{1.15em}}%
        \hspace{3pt}%
        \parbox[t]{0.82\marginparwidth}{\raggedright #1}%
    }%
}
\newcommand{\mnote}[1]{\sidenote{#1}}
\NewDocumentCommand{\chapteroverview}{O{Chapter Overview} +m}{%
    \begin{tcolorbox}[chapteroverview/base,title={#1}]
        #2
    \end{tcolorbox}
}
\makeatother
