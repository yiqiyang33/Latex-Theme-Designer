\end{homeworkProblem}

% A dotted number sets a prefix as well, for a problem set that belongs to a chapter:

\begin{homeworkProblem}[1.1][Chapter-Qualified Problem]
Explicit numbers are integers or dotted integers. The representation is chosen

\end{homeworkProblem}

% <...> overrides the heading word for this one problem only.
\begin{homeworkProblem}[Extra Credit]<Bonus>
Set the word for every problem at once with
\texttt{\textbackslash renewcommand\{\textbackslash homeworkProblemName\}\{Question\}},
the representation with \texttt{\textbackslash homeworkProblemNumbering\{Roman\}}, and
switch part numbering between \texttt{(a)} and \texttt{1.1} with
\texttt{\textbackslash homeworkSectionStandalone} or
\texttt{\textbackslash homeworkSectionNested} --- problems and sections are configured
independently. Cross-references follow an override too: this is
Problem~\ref{prob:renumbered}.
\end{homeworkProblem}
