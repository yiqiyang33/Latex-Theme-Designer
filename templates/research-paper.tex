% Import only references to other sections, avoiding duplicate local labels.
% Compile main.tex first to make these cross-section references available.
\makeatletter
\newcommand{\loadmainreferences}[1]{%
  \IfFileExists{main.aux}{\externaldocument[main-][nocite]{main}}{%
    \IfFileExists{../main.aux}{\externaldocument[main-][nocite]{../main}}{%
      \IfFileExists{../.overleaf-codex/local-build/main.aux}{%
        \externaldocument[main-][nocite]{../.overleaf-codex/local-build/main}%
      }{}%
    }%
  }%
  \@for\mainref:=#1\do{%
    \@ifundefined{r@main-\mainref}{}{%
      \expandafter\let\expandafter\mainrefvalue\csname r@main-\mainref\endcsname
      \expandafter\let\csname r@\mainref\endcsname\mainrefvalue
      \expandafter\let\expandafter\mainrefvalue\csname r@main-\mainref @cref\endcsname
      \expandafter\let\csname r@\mainref @cref\endcsname\mainrefvalue
    }%
  }%
}
\makeatother
