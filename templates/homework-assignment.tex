% The title block follows the active Toolkit style: course line, title, field labels
% and rule all take their colour from the theme, so switching preset restyles the whole
% header rather than leaving it black above coloured problem headings.
\newcommand{\makehomeworktitle}{
    \begin{center}
        {\Large\bfseries\color{theme-subsection} \homeworkCourse\par}
        \vspace{0.35em}
        {\huge\bfseries\color{theme-chapter} \homeworkTitle\par}
        \vspace{0.7em}
        {\color{theme-header-rule}\rule{0.42\linewidth}{0.7pt}\par}
        \vspace{0.7em}
        \begin{tabular}{rl}
            {\bfseries\color{theme-section}Name:} & \homeworkAuthor \\
            {\bfseries\color{theme-section}Instructor:} & \homeworkInstructor \\
            {\bfseries\color{theme-section}Due:} & \homeworkDueDate
        \end{tabular}
    \end{center}
    \vspace{1.2em}
}

% -------------------- Homework Environments --------------------
\newcounter{homeworkProblemCounter}
\newcounter{homeworkSectionCounter}[homeworkProblemCounter]
\renewcommand{\thehomeworkSectionCounter}{\alph{homeworkSectionCounter}}

\NewDocumentEnvironment{homeworkProblem}{o}
{
    \refstepcounter{homeworkProblemCounter}
    \setcounter{homeworkSectionCounter}{0}
    \section*{Problem \thehomeworkProblemCounter\IfNoValueTF{#1}{}{ --- #1}}
    \addcontentsline{toc}{section}{Problem \thehomeworkProblemCounter\IfNoValueTF{#1}{}{ --- #1}}
}
{
    \vspace{0.4em}
}

\NewDocumentEnvironment{homeworkSection}{o}
{
    \refstepcounter{homeworkSectionCounter}
    \subsection*{(\thehomeworkSectionCounter)\IfNoValueTF{#1}{}{ #1}}
}
{
    \vspace{0.25em}
}

        coltitle=note-title-fg,
        colbacktitle=note-title-bg,
        fonttitle=\bfseries,
        boxrule=0.8pt,
        arc=2pt,
        left=8pt,
        right=8pt,
        top=7pt,
        bottom=7pt
    ]
}
{
    \end{tcolorbox}
}
